$ErrorActionPreference = 'Stop'
$root = (Resolve-Path (Join-Path $PSScriptRoot '..')).Path
$changes = [System.Collections.Generic.List[object]]::new()
function CheckedPath([string]$relative) {
    $full = [IO.Path]::GetFullPath((Join-Path $root $relative))
    if (-not $full.StartsWith($root + [IO.Path]::DirectorySeparatorChar, [StringComparison]::OrdinalIgnoreCase)) { throw "Outside workspace: $relative" }
    return $full
}
function MoveLocal([string]$from, [string]$to) {
    $src = CheckedPath $from; $dst = CheckedPath $to
    if (-not (Test-Path -LiteralPath $src)) { return }
    if (Test-Path -LiteralPath $dst) { throw "Destination exists: $to" }
    New-Item -ItemType Directory -Force -Path (Split-Path $dst) | Out-Null
    Move-Item -LiteralPath $src -Destination $dst
    $changes.Add(@{action='move';from=$from;to=$to})
}
function DeleteLocal([string]$relative, [string]$reason) {
    $target = CheckedPath $relative
    if (-not (Test-Path -LiteralPath $target)) { return }
    $files = @(Get-ChildItem -LiteralPath $target -Recurse -File -Force)
    if (Test-Path -LiteralPath $target -PathType Leaf) { $files = @(Get-Item -LiteralPath $target) }
    foreach ($file in $files) {
        $changes.Add(@{action='delete';path=$file.FullName.Substring($root.Length+1);bytes=$file.Length;sha256=(Get-FileHash -LiteralPath $file.FullName -Algorithm SHA256).Hash;reason=$reason})
    }
    Remove-Item -LiteralPath $target -Recurse -Force
}
